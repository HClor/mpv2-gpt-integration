🔍 Диагностика результатов тестов пользователя

📋 Инструкция

Откройте консоль браузера (F12) и скопируйте следующий код:

// ==================================================
// ДИАГНОСТИЧЕСКИЙ КОД ДЛЯ КОНСОЛИ БРАУЗЕРА
// ==================================================
// Проверяет данные пользователя ID 2 через DOM-элементы на странице
// Запустите на страницах с сниппетами: myAchievements, myCertificates, userProfile, testHistory

(async function() {
    console.log('%c=== ДИАГНОСТИКА РЕЗУЛЬТАТОВ ТЕСТОВ ===', 'color: #569cd6; font-size: 16px; font-weight: bold;');

    const userId = 2; // ID пользователя для проверки

    // Функция для получения данных из localStorage
    function checkLocalStorage() {
        console.log('\n%c1. ПРОВЕРКА LocalStorage:', 'color: #4fc3f7; font-weight: bold;');
        console.log('---'.repeat(20));

        const keys = Object.keys(localStorage);
        const testKeys = keys.filter(k => k.includes('test') || k.includes('user') || k.includes('session'));

        if (testKeys.length > 0) {
            console.log('Найдено ключей связанных с тестами:', testKeys.length);
            testKeys.forEach(key => {
                console.log(`  📦 ${key}:`, localStorage.getItem(key));
            });
        } else {
            console.log('⚠️  Нет данных о тестах в localStorage');
        }
    }

    // Функция для проверки DOM элементов на текущей странице
    function checkPageContent() {
        console.log('\n%c2. АНАЛИЗ СОДЕРЖИМОГО СТРАНИЦЫ:', 'color: #4fc3f7; font-weight: bold;');
        console.log('---'.repeat(20));

        // Проверяем заголовок страницы
        const pageTitle = document.querySelector('h1, h2');
        console.log('Заголовок страницы:', pageTitle?.textContent || 'не найден');

        // Проверяем алерты (сообщения об ошибках или предупреждения)
        const alerts = document.querySelectorAll('.alert');
        if (alerts.length > 0) {
            console.log(`\nНайдено сообщений (alerts): ${alerts.length}`);
            alerts.forEach((alert, i) => {
                const className = alert.className;
                const text = alert.textContent.trim();
                const type = className.includes('success') ? '✅' :
                            className.includes('warning') ? '⚠️' :
                            className.includes('danger') ? '🔴' : 'ℹ️';
                console.log(`  ${type} Alert ${i + 1}: ${text.substring(0, 100)}...`);
            });
        }

        // Проверяем таблицы (testHistory)
        const tables = document.querySelectorAll('table');
        if (tables.length > 0) {
            console.log(`\nНайдено таблиц: ${tables.length}`);
            tables.forEach((table, i) => {
                const rows = table.querySelectorAll('tbody tr');
                console.log(`  📊 Таблица ${i + 1}: строк данных = ${rows.length}`);

                // Анализируем первые 3 строки
                Array.from(rows).slice(0, 3).forEach((row, rowIndex) => {
                    const cells = Array.from(row.querySelectorAll('td')).map(td => td.textContent.trim());
                    console.log(`    Строка ${rowIndex + 1}:`, cells);
                });
            });
        }

        // Проверяем карточки (certificates, achievements)
        const cards = document.querySelectorAll('.card');
        if (cards.length > 0) {
            console.log(`\nНайдено карточек: ${cards.length}`);
            Array.from(cards).slice(0, 5).forEach((card, i) => {
                const title = card.querySelector('.card-title, h5, h6')?.textContent.trim();
                const badge = card.querySelector('.badge')?.textContent.trim();
                console.log(`  🎴 Карточка ${i + 1}: ${title || 'без заголовка'} ${badge ? `[${badge}]` : ''}`);
            });
        }

        // Проверяем статистику
        const stats = document.querySelectorAll('.fw-bold.fs-4, h3');
        if (stats.length > 0) {
            console.log('\nСтатистика на странице:');
            stats.forEach((stat, i) => {
                const value = stat.textContent.trim();
                const label = stat.nextElementSibling?.textContent.trim() ||
                             stat.parentElement?.textContent.trim();
                if (value && label) {
                    console.log(`  📈 ${label}: ${value}`);
                }
            });
        }
    }

    // Функция для выполнения AJAX-запроса к API (если есть)
    async function checkAPI() {
        console.log('\n%c3. ПРОВЕРКА API:', 'color: #4fc3f7; font-weight: bold;');
        console.log('---'.repeat(20));

        // Пытаемся определить базовый URL API
        const apiEndpoints = [
            '/assets/components/testsystem/api.php',
            '/api/test-results',
            '/connector.php'
        ];

        for (const endpoint of apiEndpoints) {
            try {
                console.log(`Проверяю endpoint: ${endpoint}`);
                const response = await fetch(endpoint + '?action=getUserTests&userId=' + userId, {
                    credentials: 'same-origin'
                });

                if (response.ok) {
                    const data = await response.json();
                    console.log('✅ Ответ получен:', data);
                    return;
                } else {
                    console.log(`⚠️  Endpoint ${endpoint}: ${response.status} ${response.statusText}`);
                }
            } catch (error) {
                console.log(`❌ Ошибка при запросе к ${endpoint}:`, error.message);
            }
        }

        console.log('ℹ️  API endpoints не найдены или недоступны');
    }

    // Функция для прямого SQL-запроса через AJAX (требует специального PHP обработчика)
    async function checkDatabase() {
        console.log('\n%c4. ПРОВЕРКА БАЗЫ ДАННЫХ (через AJAX):', 'color: #4fc3f7; font-weight: bold;');
        console.log('---'.repeat(20));

        const diagnosticUrl = '/diagnostic_user_test_results.php?user_id=' + userId;

        try {
            const response = await fetch(diagnosticUrl);
            if (response.ok) {
                const text = await response.text();
                console.log('Результат диагностики из БД:');
                console.log(text);
            } else {
                console.log('⚠️  Файл diagnostic_user_test_results.php недоступен');
                console.log('   Используйте этот файл через консоль MODX');
            }
        } catch (error) {
            console.log('❌ Ошибка:', error.message);
            console.log('ℹ️  Для диагностики БД используйте консоль MODX (см. файл diagnostic_user_test_results.php)');
        }
    }

    // Выполняем все проверки
    checkLocalStorage();
    checkPageContent();
    await checkAPI();
    await checkDatabase();

    // Рекомендации
    console.log('\n%c5. РЕКОМЕНДАЦИИ:', 'color: #ffb74d; font-weight: bold;');
    console.log('---'.repeat(20));
    console.log('1. Откройте консоль MODX и выполните код из файла diagnostic_user_test_results.php');
    console.log('2. Проверьте вкладку Network в DevTools для анализа AJAX-запросов');
    console.log('3. Проверьте поля test_sessions.status, test_sessions.passed, test_sessions.score');
    console.log('4. Убедитесь, что тесты связаны с траекториями обучения (если используются)');

    console.log('\n%c=== ДИАГНОСТИКА ЗАВЕРШЕНА ===', 'color: #81c784; font-size: 16px; font-weight: bold;');
})();

🔧 Альтернативный способ (SQL через консоль MODX)

Если у вас есть доступ к консоли MODX, выполните PHP-скрипт:

// Загрузите файл diagnostic_user_test_results.php
// Или скопируйте его содержимое в консоль MODX
require_once MODX_BASE_PATH . 'diagnostic_user_test_results.php';

📊 Что проверяет диагностика

  • test_sessions: записи тестов, статус, баллы, флаг passed
  • test_learning_path_enrollments: записи на траектории
  • test_learning_path_progress: прогресс по траекториям
  • test_learning_path_step_completion: завершённые шаги
  • test_learning_path_user_achievements: достижения пользователя
  • test_certificates: выданные сертификаты

⚠️ Возможные проблемы

  • Поле test_sessions.passed не обновляется автоматически при score >= 70
  • Статус test_sessions.status остаётся 'in_progress' после завершения
  • Тест не связан с траекторией обучения (нет записи в step_completion)
  • Достижения не начисляются автоматически после прохождения теста
  • Сертификат не выдаётся, хотя траектория завершена на 100%

✅ Возможные решения

-- Обновить флаг passed для всех тестов с баллом >= 70
UPDATE modx_test_sessions
SET passed = 1
WHERE user_id = 2 AND score >= 70 AND passed = 0;

-- Обновить статус на 'completed' для завершённых тестов
UPDATE modx_test_sessions
SET status = 'completed'
WHERE user_id = 2 AND finished_at IS NOT NULL AND status != 'completed';

-- Проверить связь теста с траекторией
SELECT ts.*, lpsc.*
FROM modx_test_sessions ts
LEFT JOIN modx_test_learning_path_step_completion lpsc ON lpsc.session_id = ts.id
WHERE ts.user_id = 2;
