Script Duplication Diagnostic

Instructions:

  1. Open this file in browser OR copy script to console
  2. OR paste this into browser console on the problem page:
// Вставьте это в консоль браузера на странице с ошибкой:

const scripts = document.querySelectorAll('script');
console.log('Total scripts:', scripts.length);

const learningMaterials = Array.from(scripts).filter(s =>
    s.src && s.src.includes('learning-materials.js')
);
console.log('learning-materials.js loaded:', learningMaterials.length, 'times');
learningMaterials.forEach((s, i) => console.log(`  ${i+1}.`, s.src));

const inline = Array.from(scripts).filter(s =>
    !s.src && s.textContent.includes('materialQuillEditor')
);
console.log('Inline scripts with materialQuillEditor:', inline.length);
inline.forEach((s, i) => console.log(`  ${i+1}. Preview:`, s.textContent.substring(0, 100)));
